集成说明

// 在页面中集成抽取系统
<div id="my-container"></div>

<script src="./pk-system/core/module.js"></script>
<script>
  // 初始化系统
  const pkInstance = window.initPKModule({
    target: '#my-container',
    namespace: 'mypk-', // 可选：自定义命名空间
    autoStyleIsolation: true // 可选：自动样式隔离
  });
  
  // 销毁系统（可选）
  // pkInstance.destroy();
</script>
